async function loadFeed() { const feed = document.getElementById("feed"); try { // 1. Trouver ton compte const search = await fetch( "https://mastodon.social/api/v1/accounts/search?q=WireDistrict" ); const accounts = await search.json(); if (!accounts || accounts.length === 0) { feed.innerHTML = "Compte Mastodon introuvable ❌"; return; } const account = accounts[0]; // 2. Récupérer TES posts uniquement const postsRes = await fetch( `https://mastodon.social/api/v1/accounts/${account.id}/statuses?limit=10` ); const posts = await postsRes.json(); feed.innerHTML = ""; posts.forEach(post => { const div = document.createElement("div"); div.className = "post"; let html = ""; html += `
${post.content}
`; if (post.media_attachments && post.media_attachments.length > 0) { post.media_attachments.forEach(media => { if (media.type === "image") { html += ``; } }); } div.innerHTML = html; feed.appendChild(div); }); } catch (err) { console.log(err); feed.innerHTML = "Erreur chargement Mastodon ❌"; } } loadFeed();
